t = float(input('Введите температуру: '))

if t < -40:
    print('Экстремально холодно')
elif t <= -20:
    print('Очень холодно')
elif t <= 0:
    print('Холодно')
elif t <= 15:
    print('Прохладно')
elif t <= 25:
    print('Тепло')
elif t <= 35:
    print('Жарко')
else:
    print('Экстремально жарко')
