x = float(input("Введите точку х: "))
y = float(input("Введите точку y: "))
if x == 0 and y == 0:
    print('Начало координат')
elif y == 0:
    print('На оси Х')
elif x == 0:
    print('На оси Y')
elif x > 0 and y > 0:
    print('I четверть')
elif x < 0 and y > 0:
    print('II четверть')
elif x < 0 and y < 0:
    print('III четверть')
elif x > 0 and y < 0:
    print('IV четверть')
