n = int(input())
a = list(map(int, input().split()))

max_a = a[0]
min_a = a[0]
max_index = 0
min_index = 0

for i in range(1, n):
    if a[i] > max_a:
        max_a = a[i]
        max_index = i
    if a[i] < min_a:
        min_a = a[i]
        min_index = i

print(f"Максимум: {max_a} (индекс {max_index})")
print(f"Минимум: {min_a} (индекс {min_index})")
