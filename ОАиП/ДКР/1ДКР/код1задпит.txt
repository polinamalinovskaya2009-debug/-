import math
x = float(input("Введите значение x: "))
if x < -10:
    y = math.exp(x)
elif -10 <= x < -2:
    if abs(math.cos(x)) < 1e-9:
        print("Ошибка: деление на ноль")
    else:
        y = math.tan(x) / math.sin(x) + math.cos(2 ** x)
        print(f"y = {y:.12f}")
else:
    y = 37 + x / (0.1 ** x)
    print(f"y = {y:.12f}")
